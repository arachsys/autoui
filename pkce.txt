<!doctype html>
<title>OpenRouter PKCE demo</title>

<script type="module">
document.getElementById('auth').addEventListener('click', async () => {
  const verifier = crypto.getRandomValues(new Uint8Array(32)).toBase64(
    { alphabet: 'base64url', omitPadding: true });
  const challenge = new Uint8Array(await crypto.subtle.digest('SHA-256',
      new TextEncoder().encode(verifier))).toBase64(
    { alphabet: 'base64url', omitPadding: true });
  sessionStorage.setItem('verifier', verifier);

  const url = new URL('https://openrouter.ai/auth');
  url.searchParams.set('callback_url', location.origin + location.pathname);
  url.searchParams.set('code_challenge', challenge);
  url.searchParams.set('code_challenge_method', 'S256');
  url.searchParams.set('key_label', 'PKCE demo');
  location.assign(url);
});

const code = new URLSearchParams(location.search).get('code');
const verifier = sessionStorage.getItem('verifier');

if (code && verifier) {
  history.replaceState(null, '', location.origin + location.pathname);
  sessionStorage.removeItem('verifier');

  const response = await fetch('https://openrouter.ai/api/v1/auth/keys', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      code: code,
      code_challenge_method: 'S256',
      code_verifier: verifier
    })
  });

  if (response.ok)
    document.getElementById('key').value = (await response.json()).key;
}
</script>

<input id="key" placeholder="API key" size="72" type="text">
<button id="auth">Authorise</button>
